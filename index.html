    <!-- JavaScript: The calculator's logic -->
    <script>
        const display = document.getElementById('display');
        let currentInput = '0';
        let shouldResetDisplay = false;

        function appendNumber(num) {
            // If display shows 0 or we just calculated a result, overwrite it
            if (currentInput === '0' || shouldResetDisplay) {
                currentInput = num;
                shouldResetDisplay = false;
            } else {
                // Prevent multiple decimals in a single number sequence
                if (num === '.' && currentInput.split(/[\+\-\*\/]/).pop().includes('.')) {
                    return;
                }
                currentInput += num;
            }
            updateDisplay();
        }

        function appendOperator(op) {
            if (shouldResetDisplay) shouldResetDisplay = false;
           
            // Get the last character of the current input
            const lastChar = currentInput.slice(-1);
           
            // If the last character is an operator, replace it instead of adding another
            if (['+', '-', '*', '/'].includes(lastChar)) {
                currentInput = currentInput.slice(0, -1) + op;
            } else {
                currentInput += op;
            }
            updateDisplay();
        }

        function clearDisplay() {
            currentInput = '0';
            updateDisplay();
        }

        function calculate() {
            try {
                // Safely evaluate the mathematical string expression
                // Using Function instead of eval() is a safer practice in JS
                const result = new Function('return ' + currentInput)();
               
                // Format the output to avoid long floating-point decimals (e.g., 0.1 + 0.2)
                currentInput = Number(result.toFixed(8)).toString();
                shouldResetDisplay = true;
            } catch (error) {
                currentInput = 'Error';
                shouldResetDisplay = true;
            }
            updateDisplay();
        }

        function updateDisplay() {
            // Replace internal code symbols (* and /) with clean visual symbols for the user
            display.textContent = currentInput.replace(/\*/g, '×').replace(/\//g, '÷');
        }
    </script>
